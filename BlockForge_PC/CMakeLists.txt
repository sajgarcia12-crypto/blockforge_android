cmake_minimum_required(VERSION 3.24)
project(BlockForge_PC VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(WIN32)
  add_executable(BlockForge_PC WIN32 src/main.cpp)
  target_link_libraries(BlockForge_PC PRIVATE comdlg32)
else()
  add_executable(BlockForge_PC src/main.cpp)
endif()
add_executable(grafisript_tests tests/grafisript_tests.cpp)
enable_testing()
add_test(NAME grafisript_core COMMAND grafisript_tests)
if(MSVC)
 target_compile_options(BlockForge_PC PRIVATE /W4 /permissive-)
 target_compile_options(grafisript_tests PRIVATE /W4 /permissive-)
else()
 target_compile_options(BlockForge_PC PRIVATE -Wall -Wextra -Wpedantic)
 target_compile_options(grafisript_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
