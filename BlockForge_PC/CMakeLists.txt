cmake_minimum_required(VERSION 3.24)
project(BlockForge_PC VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(grafisript_core src/core/grafisript.hpp)
add_executable(BlockForge_PC src/main.cpp)
target_link_libraries(BlockForge_PC PRIVATE grafisript_core)
add_executable(grafisript_tests tests/grafisript_tests.cpp)
target_link_libraries(grafisript_tests PRIVATE grafisript_core)
enable_testing()
add_test(NAME grafisript_core COMMAND grafisript_tests)
if(MSVC)
 target_compile_options(grafisript_core PRIVATE /W4 /permissive-)
 target_compile_options(BlockForge_PC PRIVATE /W4 /permissive-)
else()
 target_compile_options(grafisript_core PRIVATE -Wall -Wextra -Wpedantic)
 target_compile_options(BlockForge_PC PRIVATE -Wall -Wextra -Wpedantic)
endif()
